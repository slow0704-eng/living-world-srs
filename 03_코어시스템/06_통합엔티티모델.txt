5.1 원칙
  [R-5.1] 인간, 늑대, 토끼, 관목은 모두 동일한 Entity 인터페이스를 구현한다.
          차이는 '데이터(태그와 수치)'일 뿐 '코드'가 아니다.
          -> 늑대의 사냥 로직 = 인간의 상점 방문 로직 = 같은 Utility AI 함수

  [R-5.2] 모든 엔티티는 아래 최소 공통 속성을 가진다.
          위치 / 에너지(배고픔) / 체력 / 공포도 / 포식 등급 태그 / 기억(생물만)

5.2 포식 피라미드 태그

 <표 3> 트로픽(먹이) 등급 정의
 +-------+------------------+-------------------------+------------------------+
 | 등급  | 분류             | 예시                    | 먹이 대상              |
 +-------+------------------+-------------------------+------------------------+
 | T0    | 생산자           | 관목, 수목, 풀, 이끼    | (햇빛/토양 영양)       |
 | T1    | 분해자/곤충      | 벌, 개미, 지렁이        | T0 잔해                |
 | T2    | 소형 초식/설치류 | 토끼, 다람쥐, 들쥐      | T0                     |
 | T3    | 대형 초식        | 사슴, 멧돼지, 소        | T0 (우선 관목)         |
 | T4    | 소형 육식        | 여우, 살쾡이            | T2, T1                 |
 | T5    | 대형 육식        | 늑대, 곰               | T3 우선, 없으면 T2     |
 | T6    | 지성체           | 인간, 고블린 등         | T2~T5 + 가공식품       |
 | TF    | 수생             | 소형어류 / 대형어류     | 별도 수역 체인         |
 +-------+------------------+-------------------------+------------------------+

  [R-5.3] 포식 우선순위는 '에너지 획득량 / 사냥 위험도' 비율로 결정한다.
          하드코딩된 타깃 리스트를 쓰지 않는다. (원칙 P6과 동일한 철학)

 <코드 1> 통합 엔티티 스키마 (TypeScript)
 --------------------------------------------------------------------------
 interface Entity {
   id: string;
   kind: "PLANT" | "ANIMAL" | "PERSON";
   speciesId: string;                 // "wolf", "rabbit", "human", "shrub"
   trophic: "T0"|"T1"|"T2"|"T3"|"T4"|"T5"|"T6"|"TF";

   position: { mapId: string; x: number; y: number };
   energy: number;                    // = 배고픔. 0이면 사망 (원칙 P8)
   hp: number;
   fear: number;                      // 0~100
   ageTicks: number;
   isAlive: boolean;
 }

 interface Creature extends Entity {   // ANIMAL, PERSON 공통
   stamina: StaminaState;             // 6장
   needs: Needs;                      // 6.1
   memory: MemoryEntry[];             // 8장 (동물은 '천적/위험지역'만 기록)
   relations: Record<string, Relation>;
   currentAction: ActionState | null;
 }

 interface Person extends Creature {   // 인간 전용
   name: string;
   mind: MindState;                   // 정신력 6.5
   personality: Personality;
   job: JobType;
   factionId: string | null;
   gold: number;
   inventory: ItemStack[];
   skills: Record<string, number>;
   outfit: Outfit;                    // 의복 -> 상태 태그(화려함/젖음)에 반영
   schedule: ScheduleTemplate;
 }
 --------------------------------------------------------------------------
